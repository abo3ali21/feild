#include <Wire.h>
#include <U8x8lib.h>
#include <Adafruit_Sensor.h>
#include <Adafruit_BME680.h>

#define SOIL_MOISTURE_PIN 32 // ESP32 pin D32

// Initialize OLED
U8X8_SSD1306_128X64_NONAME_HW_I2C u8x8(/* reset=*/ U8X8_PIN_NONE);

// Initialize BME680
Adafruit_BME680 bme; 

void setup() {
  Serial.begin(115200);

  u8x8.begin();
  u8x8.setFont(u8x8_font_chroma48medium8_r);
  u8x8.drawString(0, 0, "Init Sensors...");

  // Try standard Adafruit I2C address 0x77, fallback to 0x76 if needed
  if (!bme.begin(0x77)) {
    Serial.println("BME680 not found!");
    u8x8.drawString(0, 2, "BME680 Error!");
    while (1); 
  }

  bme.setTemperatureOversampling(BME680_OS_8X);
  bme.setHumidityOversampling(BME680_OS_2X);
  bme.setPressureOversampling(BME680_OS_4X);
  bme.setIIRFilterSize(BME680_FILTER_SIZE_3);
  bme.setGasHeater(320, 150);

  // Set soil moisture pin as input
  pinMode(SOIL_MOISTURE_PIN, INPUT);

  u8x8.clear();
}

void loop() {
  if (!bme.performReading()) {
    Serial.println("BME680 read failed");
    return;
  }

  // Read analog value from the soil moisture sensor (0 to 4095)
  int soilValue = analogRead(SOIL_MOISTURE_PIN);

  // Row 0: Temp & Humidity compressed to fit 16 characters
  u8x8.setCursor(0, 0);
  u8x8.print("T:"); 
  u8x8.print(bme.temperature, 1); // 1 decimal place
  u8x8.print(" H:"); 
  u8x8.print(bme.humidity, 1);
  u8x8.print("   "); // Clear trailing characters

  // Row 2: Pressure
  u8x8.setCursor(0, 2);
  u8x8.print("P:");
  u8x8.print(bme.pressure / 100.0, 1);
  u8x8.print(" hPa  ");

  // Row 4: Gas Resistance
  u8x8.setCursor(0, 4);
  u8x8.print("G:");
  u8x8.print(bme.gas_resistance / 1000.0, 1);
  u8x8.print(" kOhm ");

  // Row 6: Soil Moisture
  u8x8.setCursor(0, 6);
  u8x8.print("Soil:");
  u8x8.print(soilValue);
  u8x8.print("     "); // Clear trailing characters

  delay(2000);
}
